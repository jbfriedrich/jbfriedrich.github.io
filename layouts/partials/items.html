{{- $src := .src -}} {{- $r := partial "source.html" (dict "src" $src) -}} {{- $items := slice -}} {{- if $r.ok -}} {{- $all := partial "normalise.html" (dict "src" $src "data" $r.data) -}} {{/* Take a wider slice first: filtering after the cut can empty the tile. */}} {{- $window := int ($src.overfetch | default $src.count) -}} {{- $all = first $window $all -}} {{- $kept := slice -}} {{- range $it := $all -}} {{- $drop := false -}} {{/* .body, NOT .summary: .summary is cut to 180 runes for display, and a Mastodon cross-post puts its asides.blog link after the prose, so on a long post the link falls past the cut. Matching the display string let those items through -- and left no trace of the excluded pattern in the markup either, so a test asserting on the rendered tile passed. */}} {{- range $pat := ($src.excludes | default slice) -}} {{- if or (in $it.url $pat) (in $it.body $pat) -}} {{- $drop = true -}} {{- end -}} {{- end -}} {{- if not $drop -}}{{- $kept = $kept | append $it -}}{{- end -}} {{- end -}} {{- $items = first (int $src.count) $kept -}} {{- end -}} {{- return dict "ok" $r.ok "items" $items -}}
