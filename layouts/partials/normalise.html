{{- $src := .src -}} {{- $data := .data -}} {{- $items := slice -}} {{- if eq $src.kind "atom" -}} {{- range $e := (partial "rawlist.html" (dict "value" $data.entry)) -}} {{/* is a map when there is one, a slice when there are several */}} {{- $links := partial "rawlist.html" (dict "value" $e.link) -}} {{/* a with no attributes at all (no rel=, no href=) unmarshals to a bare string rather than a map; rawattr.html guards the -rel and -href reads, and its text content — the whole point of the bare shape — stands in for the missing -href */}} {{- $url := "" -}} {{- range $links -}} {{- if eq (partial "rawattr.html" (dict "value" . "attr" "-rel" "default" "alternate")) "alternate" -}} {{- $url = partial "rawattr.html" (dict "value" . "attr" "-href" "default" (partial "rawstring.html" (dict "value" .))) -}} {{- end -}} {{- end -}} {{- $permalink := "" -}} {{- range $links -}} {{- if eq (partial "rawattr.html" (dict "value" . "attr" "-rel" "default" "")) "related" -}} {{- $permalink = partial "rawattr.html" (dict "value" . "attr" "-href" "default" (partial "rawstring.html" (dict "value" .))) -}} {{- end -}} {{- end -}} {{- $host := "" -}} {{- if and $permalink $url -}} {{- $host = (urls.Parse $url).Host -}} {{- end -}} {{- $tags := slice -}} {{- $cats := partial "rawlist.html" (dict "value" $e.category) -}} {{- range $cats -}} {{/* a with no term= attribute unmarshals to a bare string rather than a map; rawattr.html guards the -term read, falling back to the text content itself as the tag */}} {{- $term := partial "rawattr.html" (dict "value" . "attr" "-term" "default" (partial "rawstring.html" (dict "value" .))) -}} {{- with $term -}}{{- $tags = $tags | append (lower .) -}}{{- end -}} {{- end -}} {{/* is a map when it carries a type= attribute, and a bare string when a feed omits type= (valid Atom; it defaults to "text") — rawstring.html handles either shape */}} {{- $body := partial "rawstring.html" (dict "value" $e.content) -}} {{/* is a map when a generator (WordPress, Blogger, …) emits type= on it */}} {{- $title := partial "rawstring.html" (dict "value" $e.title) -}} {{/* <updated> is the REQUIRED Atom element; <published> is optional. Fall back so a feed emitting only <updated> still dates its items, rather than silently dropping every one of them. */}} {{- $when := partial "rawstring.html" (dict "value" $e.published) -}} {{- if not $when -}}{{- $when = partial "rawstring.html" (dict "value" $e.updated) -}}{{- end -}} {{- $d := partial "safedate.html" (dict "value" $when "src" $src.key "title" $title) -}} {{- if $d.ok -}} {{/* .summary is cut to fit a card; .body is the whole thing, and is what the excludes filter in items.html matches on. They cannot be one field: matching the truncated string silently keeps any item whose excluded URL sits past the cut. */}} {{- $plain := htmlUnescape (plainify $body) -}} {{- $items = $items | append (dict "title" $title "url" $url "permalink" $permalink "host" $host "date" $d.date "tags" $tags "body" $plain "summary" (partial "truncate.html" (dict "s" $plain "n" 180)) "image" "" "rating" 0) -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq $src.kind "rss" -}} {{- range $i := (partial "rawlist.html" (dict "value" $data.channel.item)) -}} {{- $body := htmlUnescape (plainify (partial "rawstring.html" (dict "value" $i.description))) -}} {{- $title := partial "rawstring.html" (dict "value" $i.title) -}} {{/* Mastodon items have no <title>; fall back to the body */}} {{- if not $title -}}{{- $title = partial "truncate.html" (dict "s" $body "n" 140) -}}{{- end -}} {{- $tags := slice -}} {{- with $i.category -}} {{- range (partial "rawlist.html" (dict "value" .)) -}} {{- $tags = $tags | append (lower (partial "rawstring.html" (dict "value" .))) -}} {{- end -}} {{- end -}} {{- $d := partial "safedate.html" (dict "value" (partial "rawstring.html" (dict "value" $i.pubDate)) "src" $src.key "title" $title) -}} {{- if $d.ok -}} {{/* .summary is cut to fit a card; .body is the whole thing, and is what the excludes filter in items.html matches on. A Mastodon cross-post carries its asides.blog link AFTER the prose, so on a long post that link falls past the cut -- matching the truncated string kept the item and put it on the page twice. */}} {{- $items = $items | append (dict "title" $title "url" (partial "rawstring.html" (dict "value" $i.link)) "permalink" "" "host" "" "date" $d.date "tags" $tags "body" $body "summary" (partial "truncate.html" (dict "s" $body "n" 180)) "image" "" "rating" 0) -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq $src.kind "goodreads-widget" -}} {{/* $data is the widget's decoded HTML (source.html): one gr_custom_each_container_<id> block per book, in the shelf's sort order, after a header block that is dropped. The widget has no dates at all, so date is nil and the presets leave it out. The rating is the reader's own: one star_active image per star, none for an unrated book. */}} {{- range $blk := after 1 (split $data "gr_custom_each_container") -}} {{- $link := "" -}} {{- $title := "" -}} {{- with findRESubmatch `gr_custom_title[^"]*">\s*<a[^>]*href="([^"?]+)[^"]*"[^>]*>([^<]*)</a>` $blk 1 -}} {{- $link = index . 0 1 -}} {{- $title = index . 0 2 | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- end -}} {{- $author := "" -}} {{- with findRESubmatch `gr_custom_author[^"]*">\s*by\s*<a[^>]*>([^<]*)</a>` $blk 1 -}} {{- $author = index . 0 1 | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- end -}} {{/* The widget's thumbnail is tiny ("…/62032._SY75_.jpg"). The size suffix asks Goodreads' image server for a rendition, so swap it for 272px wide: twice the 136px the channels card resizes to. Not the bare original: those run to megabytes, and some are PNGs served as image/jpeg, which GetRemote rightly refuses. */}} {{- $image := "" -}} {{- with findRESubmatch `<img[^>]*\ssrc="([^"]+)"` $blk 1 -}} {{- $image = index . 0 1 | replaceRE `\._S[XY]\d+_\.` "._SX272_." -}} {{- end -}} {{- $rating := len (findRE `star_active` $blk) -}} {{- if gt $rating 5 -}}{{- $rating = 5 -}}{{- end -}} {{- if and $link $title -}} {{- $items = $items | append (dict "title" $title "url" $link "permalink" "" "host" "" "date" nil "tags" slice "body" $author "summary" $author "image" $image "rating" $rating) -}} {{- else -}} {{- warnf "friedrich: source %q has a widget entry without a title link; skipped" $src.key -}} {{- end -}} {{- end -}} {{- end -}} {{- return $items -}}
